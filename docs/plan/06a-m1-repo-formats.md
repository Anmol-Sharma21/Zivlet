```swift
/// Seeded PRNG so property tests reproduce (SplitMix64, public-domain algorithm).
public struct SplitMix64: RandomNumberGenerator, Sendable {
    private var state: UInt64
    public init(seed: UInt64) { state = seed }

    public mutating func next() -> UInt64 {
        state &+= 0x9E37_79B9_7F4A_7C15
        var z = state
        z = (z ^ (z >> 30)) &* 0xBF58_476D_1CE4_E5B9
        z = (z ^ (z >> 27)) &* 0x94D0_49BB_1331_11EB
        return z ^ (z >> 31)
    }
}
```

```swift
/// Every format Zivlet knows: 45 inputs + `html` (output only). Order = 02 §7.1.
public enum FormatID: String, CaseIterable, Codable, Sendable {
    case jpg, png, webp, heic, tiff, avif, bmp, svg, pdf
    case mp3, m4a, wav, flac, ogg, opus, aiff, wma
    case mp4, mov, mkv, webm, avi, wmv, gif
    case srt, vtt
    case docx, pages, md, rtf, rtfd, textClipping, txt, html, epub
    case pptx, ppt, xlsx, xls, ods, csv, tsv
    case zip, tar, gz, rar
}

/// Mixing family for multi-file rules (03a §2.6): GIF mixes as video, PDF and slides as document.
public enum FormatFamily: String, CaseIterable, Codable, Sendable {
    case image, audio, video, document, spreadsheet, subtitle, archive
}

/// Pair-count category (03a §3.3). The category of a pair is its source's category.
public enum FormatCategory: String, CaseIterable, Codable, Sendable {
    case raster, svg, audio, video, gif, pdf, document, slides, spreadsheet, subtitle, archive
}

public enum FormatRole: String, Codable, Sendable {
    case inputOnly, outputOnly, inputAndOutput
}
```

`Format.swift` — outline (all stored properties `let`):

```swift
public struct Format: Sendable, Hashable, Codable {
    public let id: FormatID, family: FormatFamily, category: FormatCategory, role: FormatRole
    public let extensions: [String]      // first = canonical output extension ("tar.gz" for gz)
    public let contentTypes: [String]    // first = preferred UTI
    public let isPackage: Bool           // pages, rtfd: a directory accepted as a file
    public var canonicalExtension: String { extensions[0] }
    public var wheelLabel: String { id == .gz ? "GZIP" : id.rawValue.uppercased() }
    public var isInput: Bool { role != .outputOnly }
    public var isOutput: Bool { role != .inputOnly }
    public init(_ id: FormatID, _ family: FormatFamily, _ category: FormatCategory, role: FormatRole = .inputAndOutput,
                extensions: [String], contentTypes: [String], isPackage: Bool = false)  // precondition: both non-empty
}
```

`FormatRegistry.swift` — outline:

```swift
public struct FormatRegistry: Sendable {
    public static let standard = FormatRegistry(formats: table)
    public let formats: [Format]
    // private byID: [FormatID: Format], byExtension: [String: FormatID] built in init
    public func format(_ id: FormatID) -> Format                       // preconditionFailure if the table lacks id
    public func formatID(fileName: String) -> FormatID?
    public func formatID(contentType: String, fileName: String) -> FormatID?
}
struct FileNameParts: Equatable { let stem: String; let `extension`: String? }
```

| Rule | Detail |
|---|---|
| Case | Lower-case the name before lookup (`x.JPEG`, `Report.Pages`) |
| Compound | A name ending `.tar.gz` (longer than `.tar.gz` itself) is `gz` before the single-extension lookup; `.tgz` and `.gz` are plain extensions of `gz` |
| Dot files | A leading dot is part of the stem: `.hidden` and `.png` have no extension; `trailingdot.` has none |
| UTI lookup | File-name extension wins; otherwise the UTI must match exactly one format (`org.xiph.ogg-audio` is shared by ogg and opus, so it alone returns nil) |
